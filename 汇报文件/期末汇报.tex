\begin{frame}{数据制造：官方数据诊断 $\to$ 造数据策略}
  \small
  通过与官方 benchmark 的逐 case 对比，识别出三个系统性差异，直接指导了 v2 的改进方向和 v3 的规划。

  \vspace{2mm}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{诊断发现（对比官方 benchmark）}
      \begin{itemize}
        \item 官方同一 bug 跨 test 表现不同（如 bug\_107 的 15 个 case 分叉位置和失败阶段各异），old\_fake 同类 bug 表现更均匀。
        \item 官方 debug/irq/CSR/ASSERT 信号占比远高于 old\_fake，后者几乎纯 mismatch。
        \item 官方 set1 存在争议 case（case\_5/9 sim/regr/trace 逐位相同但 gold 标注不同 bug），标签粒度更细。
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{对应造数据策略（v2 已执行，v3 规划中）}
      \begin{itemize}
        \item v2：每 bug 覆盖 2 种 test + 3 seeds，模拟跨 test 表现差异。
        \item v3：补 directed tests（debug\_basic\_test 等）+ FAILED/FATAL/ASSERT 失败入口。
        \item v3：构造同一 bug 跨 mismatch 与 FAILED 的混合 case，增加分叉时机差异。
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{技术路线：pairwise 算法 --- 从规则到学习}
  \small
  核心思路：将"直接聚类日志"转为"学习两个 case 是否同源"。评估指标本身是 pairwise BA，直接优化更有效。

  \vspace{2mm}
  \begin{columns}[T]
    \begin{column}{0.46\textwidth}
      \textbf{特征工程（294 维 pair 向量）}
      \begin{itemize}
        \item deterministic SVD64 cosine / 欧氏距离
        \item 双路 LLM embedding（features+summary）cosine
        \item token Jaccard（primary/sim/regr 分列）
        \item 结构化冲突信号：same\_primary\_signature、same\_mismatch\_type、same\_op\_pair、same\_fatal\_file 等 14 个二值特征
        \item 计数特征：num\_tokens 对数差分
      \end{itemize}
    \end{column}
    \begin{column}{0.50\textwidth}
      \textbf{5-seed half-split 对比（3 old\_fake 数据集）}
      \rowcolors{2}{ICCADLightGray}{white}
      \begin{tabularx}{\linewidth}{p{2.1cm}p{1.5cm}p{1.5cm}p{1.5cm}}
        \toprule
        \textbf{方法} & \textbf{first} & \textbf{stage2} & \textbf{stage3} \\
        \midrule
        det (SVD64) & 0.767 & 0.745 & 0.760 \\
        LLM concat & 0.785 & 0.769 & 0.768 \\
        pw\_logistic & 0.759 & 0.809 & 0.792 \\
        pw\_gbdt & 0.752 & \good{0.831} & \good{0.804} \\
        pw\_mlp & 0.757 & 0.828 & 0.797 \\
        \bottomrule
      \end{tabularx}
      \vspace{2mm}
      \smallnote{pw\_gbdt mean BA=0.796（+2.2pp）。stage2 TNR 从 0.81 提升到 0.88（+12pp）。}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{技术路线：Gated MLP --- 期中后最大架构突破}
  \small
  \textbf{GatedResidualMLP}（\texttt{pairwise\_neural\_models.py}）：在 residual backbone 前加入 feature-wise sigmoid gate。

  \vspace{2mm}
  \begin{columns}[T]
    \begin{column}{0.42\textwidth}
      \textbf{架构}
      \begin{itemize}
        \item 输入：294 维 pair 向量
        \item Gate：Sigmoid(gate(x)) $\times$ x + L1 正则
        \item Backbone：6 层残差 [512,512,512,256,256,128]
        \item GELU + LayerNorm + Dropout
        \item Focal Loss ($\gamma$=2.0)
      \end{itemize}
    \end{column}
    \begin{column}{0.52\textwidth}
      \textbf{7-dataset LODO 对比（seed-0）}
      \rowcolors{2}{ICCADLightGray}{white}
      \begin{tabularx}{\linewidth}{p{2.0cm}p{1.7cm}p{1.7cm}}
        \toprule
        \textbf{数据集} & \textbf{Residual} & \textbf{Gated} \\
        \midrule
        first & 0.7790 & \good{0.7914} \\
        stage2 & 0.7473 & \good{0.8036} \\
        stage3 & 0.7809 & \good{0.7985} \\
        VCS off. & 0.5869 & \good{0.7428} \\
        stable & 0.4880 & \good{0.6380} \\
        set1 & 0.7222 & \good{1.0000} \\
        set2 & 0.7146 & \good{0.9170} \\
        \textbf{mean} & 0.6884 & \good{\textbf{0.8130}} \\
        \bottomrule
      \end{tabularx}
    \end{column}
  \end{columns}
  \vspace{3mm}
  \normalsize
  结论：\key{Gated MLP 在所有 7 个数据集上超过 Residual MLP，0--9 seed macro BA 0.7581 vs 0.7194。}
\end{frame}

\begin{frame}{技术路线：trace.log 的使用与实验结论}
  \scriptsize
  trace log 提供执行路径证据。6 月实验用三种方式测试 trace 使用边界。

  \vspace{2mm}
  \renewcommand{\arraystretch}{1.45}
  \rowcolors{2}{ICCADLightGray}{white}
  \begin{tabularx}{\linewidth}{p{3.2cm}Xp{4.2cm}}
    \toprule
    \textbf{用法} & \textbf{关键实验数据} & \textbf{当前判断} \\
    \midrule
    Trace Route A（52d 确定性特征） & VCS +5.2\% BA / set2 $-$8.1\%。CSR/trap 受益，mismatch 受噪声干扰。 & 数据集依赖性，需 dataset-aware gate。\\
    Trace Modality Dropout & 7-domain LODO：first+5.9\%/stage2+4.5\%/set1$-$47.2\%。protection gate mean=\good{0.791}。 & 提升局部但严重破坏 set1，需 teacher/gate 保护。\\
    Anchor Trace Window & official-directed：w64 仅用 trace BA=\good{0.984} vs no\_trace 0.590。 & 对特定风格有效，适合作为诊断工具。\\
    Trace 验证/Refine & 0-9 trace refine mean BA +0.0005；veto\_boost 引入 FP 净增 235。 & 当前不启用，保留为事后复核用途。\\
    \bottomrule
  \end{tabularx}
  \renewcommand{\arraystretch}{1.18}
  \vspace{6mm}
  \normalsize
  结论：\key{trace 有信号但不稳定，不能作为全局默认特征。下一步应做 dataset-aware gating。}
\end{frame}

\begin{frame}{技术路线：completion LLM --- 专家证据抽取器}
  \small
  completion LLM 不负责聚类，不直接输出 bucket。它只做一件事：\textbf{把 sim/regr 局部信号规范化成结构化 JSON 证据}，供 selective expert 框架使用。

  \vspace{2mm}
  \begin{columns}[T]
    \begin{column}{0.44\textwidth}
      \textbf{设计约束}
      \begin{itemize}
        \item 每个 case 最多调用 1 次并缓存，不按 pair 调用
        \item 不发送 raw log lines、路径、地址、标识符
        \item 不发送 gold/meta/trace 或 bucket label
        \item 只 canonicalize 白名单枚举和布尔信号
      \end{itemize}
      \vspace{2mm}
      \textbf{输出：canonical evidence JSON}
      \begin{itemize}
        \item mechanism / stage / mismatch\_object
        \item state / trigger / evidence\_tags
      \end{itemize}
    \end{column}
    \begin{column}{0.52\textwidth}
      \textbf{第一轮实验：structured-completion（seed-0）}
      \rowcolors{2}{ICCADLightGray}{white}
      \begin{tabularx}{\linewidth}{p{2.8cm}p{1.6cm}p{1.6cm}p{1.6cm}p{1.6cm}}
        \toprule
        \textbf{方法} & \textbf{stable} & \textbf{set1} & \textbf{set2} & \textbf{mean} \\
        \midrule
        base (no LLM) & 0.565 & 0.722 & 0.775 & \good{0.687} \\
        trace only & 0.505 & 0.722 & 0.837 & 0.688 \\
        structured compl. & 0.504 & 0.458 & 0.784 & 0.582 \\
        trace + compl. & 0.565 & 0.556 & 0.634 & 0.585 \\
        \bottomrule
      \end{tabularx}
      \vspace{2mm}
      \smallnote{结论：白名单枚举太粗 → false merge 增加 → 整体未超 base。Raw-public caching 已启动，NVIDIA endpoint HTTP 429 限速。}
    \end{column}
  \end{columns}
  \vspace{2mm}
  \small
  当前判断：\key{completion 保留为 selective expert evidence，管线已通，gate 调优中。}
  \vspace{4mm}
\end{frame}

\begin{frame}{技术路线：selective expert 框架}
  \small
  \centering
  \begin{tikzpicture}[

    node distance=12mm and 14mm,
    box/.style={draw=ICCADBlue, fill=ICCADLightBlue, rounded corners=2pt,
      minimum width=32mm, minimum height=9mm, align=center, font=\scriptsize\bfseries},
    arrow/.style={-Latex, thick, draw=ICCADBlue},
    label/.style={font=\fontsize{6}{7}\selectfont\color{ICCADGray}}
  ]
    \node[box] (base) {base pair\\model};
    \node[box, right=of base] (diff) {difficulty\\score};
    \node[box, right=of diff] (expert) {trace / completion\\expert evidence};
    \node[box, below=of expert, yshift=-2mm] (gate) {validation-trained\\pair gate};
    \node[box, left=of gate] (cluster) {signed-graph\\clustering};
    \draw[arrow] (base) -- (diff) node[midway,above,label] {概率矩阵};
    \draw[arrow] (diff) -- (expert) node[midway,above,label] {高难度 case};
    \draw[arrow] (expert) -- (gate) node[midway,right,label] {专家概率};
    \draw[arrow] (gate) -- (cluster) node[midway,below,label] {融合概率};
  \end{tikzpicture}

  \vspace{1mm}
  \begin{columns}[T]
    \begin{column}{0.45\textwidth}
      \textbf{流程说明}
      \begin{enumerate}
        \setlength{\itemsep}{0pt}
        \item base model 输出 pair 概率矩阵
        \item difficulty score 筛选最高难度 $\sim$15\% 的 case
        \item 仅对高难 case 调用 trace/completion 专家
        \item gate 融合 base 与 expert 概率
        \item signed-graph 聚类产生最终 bucket
      \end{enumerate}
    \end{column}
    \begin{column}{0.50\textwidth}
      \textbf{关键约束}
      \begin{itemize}
        \setlength{\itemsep}{0pt}
        \item 专家不可用时 gate 权重严格为 0，base probability 保持不变
        \item gate 仅从 held-out training pairs 学习，不读取目标 gold 标签
        \item 每个 episode 专家 case 最少 2 个、最多 20 个
        \item completion 缺配置/HTTP 错误/rate limit 时不阻断管线
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}
